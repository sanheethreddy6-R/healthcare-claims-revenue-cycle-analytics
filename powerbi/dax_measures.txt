-- Create these measures in Power BI after loading the CSV tables.

Total Claims = COUNTROWS(Claims)

Total Billed = SUM(Claims[billed_amount])

Total Allowed = SUM(Claims[allowed_amount])

Total Collected = SUM(Claims[total_collected])

Outstanding AR = SUM(Claims[outstanding_amount])

Denied Claims =
CALCULATE(
    [Total Claims],
    Claims[claim_status] = "Denied"
)

Denial Rate = DIVIDE([Denied Claims], [Total Claims])

Paid Claims =
CALCULATE(
    [Total Claims],
    Claims[claim_status] = "Paid"
)

Paid Claim Rate = DIVIDE([Paid Claims], [Total Claims])

Gross Collection Rate = DIVIDE([Total Collected], [Total Billed])

Net Collection Rate = DIVIDE([Total Collected], [Total Allowed])

Contractual Adjustments = SUM(Claims[contractual_adjustment])

Contractual Adjustment Rate = DIVIDE([Contractual Adjustments], [Total Billed])

Denial Recoveries = SUM(Claims[recovery_amount])

Average Days to Payment = AVERAGE(Claims[days_to_payment])

Average Billed per Claim = DIVIDE([Total Billed], [Total Claims])

Average Collected per Claim = DIVIDE([Total Collected], [Total Claims])

Open Claims =
CALCULATE(
    [Total Claims],
    Claims[outstanding_amount] > 0
)

Open AR 90+ =
CALCULATE(
    [Outstanding AR],
    Claims[ar_bucket] IN {"91-120", "120+"}
)

Open AR 90+ % = DIVIDE([Open AR 90+], [Outstanding AR])

-- Date table
Date =
CALENDAR(
    MIN(Claims[service_date]),
    MAX(Claims[service_date])
)

-- Calculated columns for the Date table
Year = YEAR('Date'[Date])
Month Number = MONTH('Date'[Date])
Month = FORMAT('Date'[Date], "MMM")
Year Month = FORMAT('Date'[Date], "YYYY-MM")
